## 1. 新增文件 `js/site-resolver.js`（经典脚本，SW 用 importScripts、popup 用 <script>）

```js
/*
 * site-resolver.js —— 「页面解析旁路」站点判定（与现有嗅探完全独立）
 * 经典脚本（非 ES Module）：Service Worker 用 importScripts 引入，popup 用 <script> 引入，
 * 都会挂到 self.SiteResolver。命中下表站点时，不依赖抓到的媒体地址，
 * 而是把「当前页面 URL」交给本地 yt-dlp（方案1）或 GD3（方案2）。
 */
(function (root) {
  'use strict';

  // 命中表：yt-dlp 支持数百站点，后续要支持 Twitter/X、B站等只往这里加规则。
  // test 对 hostname 做后缀匹配（www.youtube.com / m.youtube.com 都能命中）。
  const SITES = [
    { test: /(^|\.)youtube\.com$/i,          name: 'YouTube' },
    { test: /(^|\.)youtu\.be$/i,             name: 'YouTube' },
    { test: /(^|\.)youtube-nocookie\.com$/i, name: 'YouTube' },
    // { test: /(^|\.)x\.com$/i,       name: 'X(Twitter)' },
    // { test: /(^|\.)twitter\.com$/i, name: 'X(Twitter)' },
  ];

  function safeHost(url) {
    try { return new URL(url).hostname.toLowerCase(); } catch (e) { return ''; }
  }
  function match(url) {
    const host = safeHost(url || '');
    if (!host) return null;
    return SITES.find((s) => s.test.test(host)) || null;
  }
  // YouTube 默认只下当前一条：去掉 list= 播放列表，避免一次拉整个列表
  function normalizePageUrl(url) {
    try {
      const u = new URL(url);
      const isYt = /(^|\.)youtube\.com$/i.test(u.hostname);
      const v = u.searchParams.get('v');
      if (isYt && v) return u.protocol + '//' + u.hostname + '/watch?v=' + encodeURIComponent(v);
      return u.toString();
    } catch (e) { return url; }
  }

  root.SiteResolver = { SITES, match, normalizePageUrl };
})(typeof self !== 'undefined' ? self : globalThis);
```

### 2.5 新消息 case（插在 `case 'sendToAria2'` 整块之后，与其它 case 平级）

> **精确插入点（v0.3.3 基线，已 grep 复核）**：`case 'sendToAria2': {` 在 **第 1541 行**；
> 块内 `return true;` 在 **第 1577 行**、`}` 在 **第 1578 行**；下一个 case 是 `case 'download'`（**第 1580 行**）。
> **插入位置 = 1578 之后、1580 之前**。搜不到时用特征 `case 'sendToAria2'` / `case 'download'` 重新定位。
>
> **V2.0 约束提醒**：本分支不新增下载/预览路径，故**不涉及 DNR**；若将来要注入 Referer，
> 必须显式传 `purpose` 并用 `releaseRule(url, purpose)` 释放（见 02 文档 §1.7）。
> 本分支也**不受 `captureEnabled` 暂停影响**（与 `getResources` 一致，见 02 文档 §1.8）。
```js
// ===== 页面解析旁路（YouTube 等）：不经过抓包，直接交「页面URL」=====
case 'resolvePage': {
  const pageUrl = (self.SiteResolver ? SiteResolver.normalizePageUrl(msg.url || '') : (msg.url || ''));
  if (self.SiteResolver && !SiteResolver.match(pageUrl)) {
    sendResponse({ ok: false, error: 'unsupported_site' }); break;
  }
  const mode = msg.mode === 'gd3' ? 'gd3' : 'helper';

  (async () => {
    // 取当前站点登录 Cookie（需要 manifest 声明 "cookies"；未声明则为空，公开视频不受影响）
    let cookie = msg.cookie || '';
    if (!cookie && chrome.cookies && chrome.cookies.getAll) {
      try {
        const host = new URL(pageUrl).hostname.replace(/^www\./, '');
        const list = await chrome.cookies.getAll({ domain: host });
        cookie = list.map((c) => c.name + '=' + c.value).join('; ');
      } catch (e) { /* ignore */ }
    }

    if (mode === 'gd3') {
      // —— 方案2：复用 GD3 的 aria2 兼容 RPC，把【页面URL】当 uri 下发 ——
      const rpc = aria2Rpc;
      if (!rpc) return sendResponse({ ok: false, error: 'no_rpc' });
      let endpoint = rpc, token = '';
      try {
        const u = new URL(rpc);
        if (u.searchParams.has('token')) {
          token = u.searchParams.get('token') || '';
          u.searchParams.delete('token');
          endpoint = u.toString();
        }
      } catch (e) { /* ignore */ }
      const header = ['Referer: ' + pageUrl];
      if (cookie) header.unshift('Cookie: ' + cookie);
      const options = { referer: pageUrl, header };
      const params = token ? [token, [pageUrl], options] : [[pageUrl], options];
      try {
        const resp = await fetch(endpoint, {
          method: 'POST', headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ jsonrpc: '2.0', id: String(Date.now()), method: 'aria2.addUri', params }),
        });
        const data = await resp.json();
        if (data && data.error) sendResponse({ ok: false, error: 'rpc_error', message: data.error.message });
        else sendResponse({ ok: true, gid: data && data.result });
      } catch (e) {
        sendResponse({ ok: false, error: 'network', message: String(e && e.message || e) });
      }
      return;
    }

    // —— 方案1：自研本地 yt-dlp 助手 ——
    const base = (ytHelperUrl || '').replace(/\/+$/, '');
    if (!base) return sendResponse({ ok: false, error: 'no_helper' });
    try {
      const resp = await fetch(base + '/download', {
        method: 'POST',
        headers: Object.assign(
          { 'Content-Type': 'application/json' },
          ytHelperToken ? { 'x-helper-token': ytHelperToken } : {}
        ),
        body: JSON.stringify({ url: pageUrl, cookie, playlist: !!msg.playlist, dir: msg.dir || '' }),
      });
      const data = await resp.json();
      sendResponse(data);
    } catch (e) {
      sendResponse({ ok: false, error: 'network', message: String(e && e.message || e) });
    }
  })();

  return true; // 异步响应
}
```

function pollHelperJob(jobId, statusEl) {
  const base = 'http://127.0.0.1:17890'; // 与 storage 的 ytHelperUrl 保持一致
  const timer = setInterval(async () => {
    try {
      const r = await fetch(base + '/status?id=' + encodeURIComponent(jobId));
      const d = await r.json();
      if (!d.ok) return;
      const j = d.job;
      statusEl.textContent = j.status + ' ' + (j.percent || 0).toFixed(1) + '% ' + (j.speed || '');
      if (j.status === 'completed') { clearInterval(timer); statusEl.textContent = t('rs_done', '已完成：$1', [j.out || '']); }
      if (j.status === 'error')     { clearInterval(timer); statusEl.textContent = t('rs_error', '出错：$1', [j.error]); }
    } catch (e) { /* 助手未启动 */ }
  }, 1200);
}

  const b1 = document.getElementById('rsDownloadBtn');

  if (b2) b2.addEventListener('click', () => resolveCurrentPage(true));

## 4. 方案1 本地助手 `yt_helper.cjs`（Node 内置模块，零 npm 依赖）

> 放在扩展包**外**（如 `D:\05catch\yt-helper\`），同级建 `bin/` 放 yt-dlp.exe、ffmpeg.exe，输出默认 `./downloads`。

```js
#!/usr/bin/env node
'use strict';
// 极简 yt-dlp 本地助手：仅用 Node 内置模块，无需 npm install。
// 接口：GET /ping；POST /download；GET /status?id=；GET /jobs；POST /cancel。
const http = require('http');
const { spawn } = require('child_process');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const PORT = Number(process.env.HELPER_PORT || 17890);
const TOKEN = process.env.HELPER_TOKEN || '';           // 设置后请求需带 x-helper-token
const GLOBAL_PROXY = process.env.HELPER_PROXY || '';
const DOWNLOAD_DIR = process.env.HELPER_DIR || path.join(__dirname, 'downloads');
const BIN_DIR = path.join(__dirname, 'bin');
fs.mkdirSync(DOWNLOAD_DIR, { recursive: true });

function resolveBin(name) {
  const exe = process.platform === 'win32' ? name + '.exe' : name;
  const local = path.join(BIN_DIR, exe);
  return fs.existsSync(local) ? local : name; // 没有随包就用 PATH
}
const YTDLP = resolveBin('yt-dlp');
const FFMPEG_LOCATION = fs.existsSync(BIN_DIR) ? BIN_DIR : '.';
const jobs = new Map();

function tokenOk(req) { return !TOKEN || req.headers['x-helper-token'] === TOKEN; }
function json(res, code, obj) {
  res.writeHead(code, {
    'Content-Type': 'application/json; charset=utf-8',
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'content-type,x-helper-token',
    'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
  });
  res.end(JSON.stringify(obj));
}
function readBody(req) {
  return new Promise((resolve, reject) => {
    let data = '';
    req.on('data', (c) => { data += c; if (data.length > 2 * 1024 * 1024) { req.destroy(); reject(new Error('body too large')); } });
    req.on('end', () => { try { resolve(data ? JSON.parse(data) : {}); } catch (e) { reject(e); } });
    req.on('error', reject);
  });
}

function startJob(opts) {
  const url = String(opts.url || '').trim();
  if (!/^https?:/i.test(url)) throw new Error('bad url');
  const id = crypto.randomBytes(5).toString('hex');
  const outDir = opts.dir || DOWNLOAD_DIR;
  fs.mkdirSync(outDir, { recursive: true });

  const args = [
    '--newline', '--no-color', '--verbose-progress',
    '--ffmpeg-location', FFMPEG_LOCATION,
    '-f', opts.format || 'bv*+ba/b',
    '--merge-output-format', 'mp4',
    '-o', path.join(outDir, '%(title).180B [%(id)s].%(ext)s'),
    '--windows-filenames',
    '--retries', '10', '--fragment-retries', '10',
    '--concurrent-fragments', String(opts.concurrent || 4),
  ];
  if (!opts.playlist) args.push('--no-playlist'); else args.push('--yes-playlist');
  if (opts.cookie) args.push('--add-header', 'Cookie:' + String(opts.cookie));
  const proxy = opts.proxy || GLOBAL_PROXY;
  if (proxy) args.push('--proxy', proxy);
  args.push(url);

  const job = { id, url, status: 'starting', percent: 0, speed: '', eta: '',
                title: '', out: '', error: '', logs: [], ts: Date.now() };
  jobs.set(id, job);

  let proc;
  try { proc = spawn(YTDLP, args, { windowsHide: true }); }
  catch (e) { job.status = 'error'; job.error = 'spawn failed: ' + e.message; return job; }
  job.pid = proc.pid;

  const ingest = (buf) => {
    buf.toString().split(/\r?\n/).forEach((line) => {
      if (!line) return;
      job.logs.push(line); if (job.logs.length > 500) job.logs.shift();
      let m;
      if ((m = line.match(/\[download\]\s+([\d.]+)%/))) { job.status = 'downloading'; job.percent = parseFloat(m[1]) || 0; }
      if ((m = line.match(/([\d.]+\s*[KMG]?i?B\/s)/))) job.speed = m[1];
      if ((m = line.match(/\bETA\s+([\d:]+)/))) job.eta = m[1];
      if ((m = line.match(/\[(?:Merger|ExtractAudio|VideoConvertor)\][^\n"]*?["'](.+?)["']/))) job.out = m[1];
      if ((m = line.match(/\[download\] Destination:\s*(.+)$/))) job.out = m[1].trim();
    });
  };
  proc.stdout.on('data', ingest);
  proc.stderr.on('data', ingest);
  proc.on('error', (e) => { job.status = 'error'; job.error = String(e.message || e); });
  proc.on('close', (code) => {
    if (job.status === 'error') return;
    if (code === 0) { job.status = 'completed'; job.percent = 100; job.speed = ''; }
    else { job.status = 'error'; job.error = 'yt-dlp exited with ' + code; }
  });
  return job;
}
function publicJob(j) { const { logs, ...rest } = j; return { ...rest, logTail: (logs || []).slice(-12) }; }

const server = http.createServer(async (req, res) => {
  try {
    if (req.method === 'OPTIONS') return json(res, 204, {});
    const u = new URL(req.url, 'http://127.0.0.1');
    if (u.pathname === '/ping')
      return json(res, 200, { ok: true, engine: YTDLP,
        runningJobs: [...jobs.values()].filter(j => j.status === 'downloading' || j.status === 'starting').length });
    if (!tokenOk(req)) return json(res, 401, { ok: false, error: 'bad token' });

    if (req.method === 'POST' && u.pathname === '/download') {
      const body = await readBody(req);
      const job = startJob(body);
      return json(res, 200, { ok: job.status !== 'error', jobId: job.id, error: job.error || undefined });
    }
    if (req.method === 'GET' && u.pathname === '/status') {
      const j = jobs.get(u.searchParams.get('id') || '');
      if (!j) return json(res, 404, { ok: false, error: 'no such job' });
      return json(res, 200, { ok: true, job: publicJob(j) });
    }
    if (req.method === 'POST' && u.pathname === '/cancel') {
      const body = await readBody(req);
      const j = jobs.get(body.id || '');
      if (j && !['completed', 'error'].includes(j.status)) { try { process.kill(j.pid); } catch (e) {} j.status = 'paused'; }
      return json(res, 200, { ok: true });
    }
    if (req.method === 'GET' && u.pathname === '/jobs')
      return json(res, 200, { ok: true, jobs: [...jobs.values()].slice(-30).map(publicJob) });
    return json(res, 404, { ok: false, error: 'not found' });
  } catch (e) {
    return json(res, 500, { ok: false, error: String(e && e.message || e) });
  }
});
server.listen(PORT, '127.0.0.1', () => {
  console.log('[yt-helper] listening at http://127.0.0.1:' + PORT);
  console.log('[yt-helper] yt-dlp =', YTDLP, '| ffmpeg =', FFMPEG_LOCATION, '| save =', DOWNLOAD_DIR);
});
```
